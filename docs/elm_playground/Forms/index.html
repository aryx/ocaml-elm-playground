Module Forms

type kind =
  1. | Text
  2. | Password
  3. | Checkbox
  4. | Radio
  5. | Submit
  6. | Reset
  7. | Button
  8. | Hidden
  9. | Select of (string * string) list
  10. | Textarea
type value = {
  1. text : string;
  2. checked : bool;
  3. selected : int;
}
type control = {
  1. element : Dom.element;
  2. kind : kind;
  3. name : string option;
  4. initial : value;
}
type form = {
  1. action : string;
  2. post : bool;
  3. controls : control list;
}
val control : Dom.element -> control option
val label : control -> string
val forms : Dom.element -> form list
val form_of : form list -> Dom.element -> form option
val submission : form -> value:(Dom.element -> value) -> submitter:Dom.element option -> (string * string) list
