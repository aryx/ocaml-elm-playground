Module St_ast

type pos = int * int
type literal =
  1. | L_int of int
  2. | L_large of bool * int list
  3. | L_float of float
  4. | L_char of char
  5. | L_string of string
  6. | L_symbol of string
  7. | L_array of literal list
  8. | L_nil
  9. | L_true
  10. | L_false
type expr = {
  1. e : desc;
  2. pos : pos;
}
and desc =
  1. | Lit of literal
  2. | Var of string
  3. | Assign of string * expr
  4. | Send of expr * string * expr list
  5. | Cascade of expr * (string * expr list * pos) list
  6. | Block of string list * string list * stmt list
and stmt =
  1. | Expr of expr
  2. | Return of expr * pos
type method_ = {
  1. selector : string;
  2. args : string list;
  3. temps : string list;
  4. primitive : int option;
  5. body : stmt list;
}
val arity : string -> int
val show_expr : expr -> string
val show_stmt : stmt -> string
val show_literal : literal -> string
