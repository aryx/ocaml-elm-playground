Module Mpeg_audio_header

type version =
  1. | Mpeg1
  2. | Mpeg2
  3. | Mpeg2_5
type mode =
  1. | Stereo
  2. | Joint_stereo
  3. | Dual_channel
  4. | Mono
type t = {
  1. version : version;
  2. layer : int;
  3. crc : bool;
  4. bitrate : int;
  5. sample_rate : int;
  6. padding : bool;
  7. mode : mode;
  8. mode_extension : int;
  9. channels : int;
  10. length : int;
  11. samples : int;
}
val parse : string -> int -> t option
val first_frame : string -> (int * t) option
val at_start : string -> t option
val frames : string -> (int * t) list
val version_name : version -> string
