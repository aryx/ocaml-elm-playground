type rgb = float * float * float
type light =
  1. | Sun of {
    1. towards : Vec3.t;
    2. color : rgb;
    }
  2. | Lamp of {
    1. position : Vec3.t;
    2. radius : float;
    3. color : rgb;
    }
  3. | Spot of {
    1. position : Vec3.t;
    2. aim : Vec3.t;
    3. angle : float;
    4. falloff : float;
    5. color : rgb;
    }
type scene = {
  1. camera : Camera.t;
  2. solids : Solid.t list;
  3. lights : light list;
  4. ambient : float;
  5. background : int;
}
type algorithm =
  1. | Ray_casting
  2. | Lambert
  3. | Shadow_rays
  4. | Whitted
  5. | Soft_shadows
  6. | Path_tracing

val latest : algorithm

type acceleration =
  1. | Brute_force
  2. | Bvh of Bvh.split

  1. algorithm : algorithm;
  2. epsilon : float;
  3. acceleration : acceleration;
  4. depth : int;
  5. cutoff : float;
  6. samples : int;
  7. seed : int;
}
