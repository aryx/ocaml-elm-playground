Module Scheme

type t =
  1. | Int of int
  2. | Real of float
  3. | Bool of bool
  4. | Char of int
  5. | Str of string
  6. | Sym of string
  7. | Nil
  8. | Pair of t * t
  9. | Vector of t array
  10. | Struct of string * t list
  11. | Image of Scheme_image.t
  12. | Proc of proc
  13. | Void
and proc =
  1. | Prim of string
  2. | Closure of lambda * env
  3. | Cont of kont
  4. | Make of string * int
  5. | Get of string * int * string
  6. | Is of string
and loc = int
and env = (string * loc) list
and expr = {
  1. desc : desc;
  2. span : Sexpr.span;
}
and desc =
  1. | Quote of t
  2. | Var of string
  3. | Lambda of lambda
  4. | If of expr * expr * expr
  5. | Set of string * expr
  6. | App of expr * expr list
  7. | Seq of expr list
  8. | Define of string * expr
  9. | Define_struct of string * string list
  10. | Big_bang of expr * (string * expr) list
and lambda = {
  1. params : string list;
  2. rest : string option;
  3. locals : string list;
  4. body : expr list;
  5. name : string;
}
and kont =
  1. | Halt
  2. | K_if of expr * expr * env * kont
  3. | K_app of t list * expr list * env * Sexpr.span * kont
  4. | K_set of loc * kont
  5. | K_seq of expr list * env * kont
  6. | K_define of string * kont
  7. | K_big_bang of t list * (string * expr) list * string list * env * Sexpr.span * kont
type style =
  1. | Write
  2. | Constructor
val print : style -> t -> string
val display : t -> string
val list : t list -> t
val to_list : t -> t list option
val truthy : t -> bool
val equal : t -> t -> bool
val kind : t -> string
