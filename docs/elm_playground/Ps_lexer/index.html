Module Ps_lexer

type kind =
  1. | Int of int
  2. | Real of float
  3. | Name of string
  4. | Literal of string
  5. | String of string
  6. | Open_brace
  7. | Close_brace
type token = {
  1. kind : kind;
  2. start : int;
  3. stop : int;
}
val next : string -> int -> (token, string) Stdlib.result option
val tokens : string -> (kind list, string) Stdlib.result
