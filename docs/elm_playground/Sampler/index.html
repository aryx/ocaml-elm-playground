Module Sampler

type sample = {
  1. data : Signal.t;
  2. root : int;
}
type direction =
  1. | Forward
  2. | Backward
type loop =
  1. | Forever
  2. | Until_release
  3. | Off
type settings = {
  1. start : float;
  2. loop_start : float;
  3. loop_end : float;
  4. end_ : float;
  5. direction : direction;
  6. tune : float;
  7. crossfade : float;
  8. gain : float;
  9. loop : loop;
  10. release : float option;
}
val default : settings
val voice : sample -> settings -> key:int -> velocity:float -> Polyphony.voice
val seconds : sample -> settings -> int -> float
type play =
  1. | Key
  2. | Oneshot
  3. | Mute_group
  4. | Loop
type pad = {
  1. sample : sample;
  2. settings : settings;
  3. play : play;
  4. pan : float;
}
val pad : ?settings:settings -> ?play:play -> ?pan:float -> sample -> pad
type kit
val kit : ?base:int -> pad array -> kit
val press : kit -> int -> float -> unit
val release : kit -> int -> unit
val fill : kit -> Signal.stereo -> unit
val sounding : kit -> int
