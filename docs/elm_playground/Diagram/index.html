type kind =
  1. | Box
  2. | Connector
type ends =
  1. | Begin
  2. | End
type master = {
  1. name : string;
  2. kind : kind;
  3. cells : (string * string) list;
  4. geometry : (string * string) list;
  5. points : (string * string * Ortho_route.dir) list;
}
type shape = {
  1. id : int;
  2. master : string;
  3. kind : kind;
  4. rows : (string * string) list;
  5. geometry : (string * string) list;
  6. points : (string * string * Ortho_route.dir) list;
  7. text : string;
  8. glue : (ends * int * int) list;
}
