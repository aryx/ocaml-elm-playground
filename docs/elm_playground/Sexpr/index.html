type span = {
  1. start : int;
  2. stop : int;
}
type t = {
  1. datum : datum;
  2. span : span;
}
and datum =
  1. | Int of int
  2. | Float of float
  3. | Str of string
  4. | Sym of string
  5. | Char of int
  6. | Bool of bool
  7. | List of t list * t option
  8. | Vector of t list
val make : datum -> span -> t
val sym : string -> span -> t
val nowhere : span
val to_string : t -> string
