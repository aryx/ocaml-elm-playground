Module Recur

type freq =
  1. | Daily
  2. | Weekly
  3. | Monthly
  4. | Yearly
type until =
  1. | Until_date of Civil.date
  2. | Until_time of Civil.date * int
type rule = {
  1. freq : freq;
  2. interval : int;
  3. by_day : int list;
  4. by_month_day : int list;
  5. week_start : int;
  6. count : int option;
  7. until : until option;
}
val make : freq -> rule
val occurrences : ?at:int -> rule -> start:Civil.date -> from:Civil.date -> upto:Civil.date -> Civil.date list
