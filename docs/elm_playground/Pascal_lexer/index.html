Module Pascal_lexer

type kind =
  1. | Keyword of string
  2. | Name of string
  3. | Int of int
  4. | Char of char
  5. | String of string
  6. | Symbol of string
  7. | Eof
type token = {
  1. kind : kind;
  2. line : int;
  3. col : int;
}
exception Error of int * int * string
val tokens : string -> token list
val keywords : string list
val show : kind -> string
