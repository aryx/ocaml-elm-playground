type pattern =
  1. | Plain
  2. | Checker of int * float
  3. | Marble of int * float
  4. | Wood of int * float
  5. | Solid_function of Vec3.t -> int
  6. | Uv_function of u:float -> v:float -> int

  1. color : int;
  2. pattern : pattern;
  3. material : Material.t;
}

type primitive =
  1. | Ball
  2. | Cube
  3. | Cylinder
  4. | Cone
  5. | Torus of float
  6. | Half_space
type t =
  1. | Sphere of Vec3.t * float * surface
  2. | Plane of Vec3.t * float * surface
  3. | Triangle of {
    1. points : Vec3.t * Vec3.t * Vec3.t;

    3. uvs : (float * float) * (float * float) * (float * float);
    4. surface : surface;
    }

    1. primitive : primitive;
    2. transform : Transform.t;
