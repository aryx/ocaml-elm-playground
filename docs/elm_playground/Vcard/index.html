type name = {
  1. family : string;
  2. given : string;
  3. additional : string;
  4. prefix : string;
  5. suffix : string;
}
type phone = {
  1. number : string;
  2. kinds : string list;
}
type email = {
  1. address : string;
  2. kinds : string list;
}
type address = {
  1. street : string;
  2. locality : string;
  3. region : string;
  4. code : string;
  5. country : string;
  6. kinds : string list;
}
type card = {
  1. uid : string;
  2. name : name;
  3. full_name : string;
  4. org : string;
  5. title : string;
  6. phones : phone list;
  7. emails : email list;
  8. addresses : address list;
  9. birthday : Civil.date option;
  10. note : string;
}
