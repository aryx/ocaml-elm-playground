Module Formula

type cell = int * int
type expr =
  1. | Number of float
  2. | Ref of cell
  3. | Range of cell * cell
  4. | Unary of char * expr
  5. | Binop of char * expr * expr
  6. | Call of string * expr list
type content =
  1. | Formula of expr
  2. | Invalid of string
  3. | Value of float
  4. | Text of string
  5. | Blank
val content_of : string -> content
val parse : string -> (expr, string) Stdlib.result
val refs : expr -> cell list
val name_of_cell : cell -> string
val cell_of_name : string -> cell option
val to_string : expr -> string
val shift : (int * int) -> expr -> expr
