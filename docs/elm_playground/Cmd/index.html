Module Cmd

type http_error =
  1. | Bad_url of string
  2. | Timeout
  3. | Network_error of string
  4. | Bad_status of int
  5. | Bad_body of string
type http_response = {
  1. url : string;
  2. status : int;
  3. headers : (string * string) list;
  4. body : string;
}
type 'msg t =
  1. | None
  2. | Msg of 'msg
  3. | Http_get of Cap.network * string * (http_response, http_error) Stdlib.result -> 'msg
  4. | Http_post of Cap.network * string * string * string * (http_response, http_error) Stdlib.result -> 'msg
  5. | Batch of 'msg t list
val none : 'msg t
val batch : 'msg t list -> 'msg t
val to_list : 'msg t -> 'msg t list
