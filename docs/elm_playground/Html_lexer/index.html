Module Html_lexer

type attribute = string * string
type token =
  1. | Doctype of string
  2. | Start_tag of {
    1. name : string;
    2. attributes : attribute list;
    3. extensions : attribute list;
    4. origin : Dtd.origin;
    5. self_closing : bool;
    }
  3. | End_tag of string
  4. | Text of string
  5. | Comment of string
val tokenize : string -> token list
val to_string : token -> string
val attribute : string -> attribute list -> string option
